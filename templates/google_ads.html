Setup: Google Ads Script (one time)
  1. In Google Ads: Tools → Bulk actions → Scripts → + → New script.
  2. Delete everything in the editor, paste the script below, and click Authorize.
  3. Click Preview once (should log "Lister: ok"), then Save and Run.
  4. Set the script's Frequency to Hourly.
// Lister — Google Ads daily spend push
var LISTER_URL = '{{ ingest_url }}';
var LISTER_TOKEN = '{{ ingest_token }}';

function main() {
  var tz = AdsApp.currentAccount().getTimeZone();
  var now = new Date();
  // Lister says how many days to send: 60 for the first run / once a day, otherwise 3
  var plan = JSON.parse(UrlFetchApp.fetch(LISTER_URL, {headers: {'X-Ingest-Token': LISTER_TOKEN}}).getContentText());
  var days = plan.days || 3;
  var dates = [];
  for (var i = 0; i < days; i++) {
    dates.push(Utilities.formatDate(new Date(now.getTime() - i * 86400000), tz, 'yyyy-MM-dd'));
  }
  var from = dates[dates.length - 1], to = dates[0];

  var products = [];
  var it = AdsApp.search(
    "SELECT segments.date, campaign.id, campaign.name, segments.product_item_id, segments.product_title, " +
    "metrics.impressions, metrics.clicks, metrics.cost_micros, metrics.conversions, metrics.conversions_value " +
    "FROM shopping_performance_view WHERE segments.date BETWEEN '" + from + "' AND '" + to + "' AND metrics.impressions > 0");
  while (it.hasNext()) {
    var r = it.next();
    products.push({date: r.segments.date, campaign_id: r.campaign.id, campaign: r.campaign.name,
      item_id: r.segments.productItemId, title: r.segments.productTitle,
      impressions: r.metrics.impressions, clicks: r.metrics.clicks, cost_micros: r.metrics.costMicros,
      conversions: r.metrics.conversions, conv_value: r.metrics.conversionsValue});
  }

  var campaigns = [];
  var it2 = AdsApp.search(
    "SELECT segments.date, campaign.id, campaign.name, campaign.advertising_channel_type, " +
    "metrics.impressions, metrics.clicks, metrics.cost_micros, metrics.conversions, metrics.conversions_value " +
    "FROM campaign WHERE segments.date BETWEEN '" + from + "' AND '" + to + "' AND metrics.impressions > 0");
  while (it2.hasNext()) {
    var c = it2.next();
    campaigns.push({date: c.segments.date, campaign_id: c.campaign.id, campaign: c.campaign.name,
      channel: c.campaign.advertisingChannelType,
      impressions: c.metrics.impressions, clicks: c.metrics.clicks, cost_micros: c.metrics.costMicros,
      conversions: c.metrics.conversions, conv_value: c.metrics.conversionsValue});
  }

  var resp = UrlFetchApp.fetch(LISTER_URL, {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    headers: {'X-Ingest-Token': LISTER_TOKEN},
    payload: JSON.stringify({dates: dates, products: products, campaigns: campaigns})
  });
  if (resp.getResponseCode() !== 200) {
    throw new Error('Lister push failed: ' + resp.getResponseCode() + ' ' + resp.getContentText().slice(0, 300));
  }
  Logger.log('Lister: ok ' + resp.getContentText());
}
